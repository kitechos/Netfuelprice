        function updateRowLabels(rowId) {
            const row = rows.find(r => r.id === rowId);
            if (!row) return;
            const region = taxRates[row.regionKey];
            const unitSpan = document.getElementById(`unit-${rowId}`);

                unitSpan.textContent = region.type === 'CA' ? 'CAD/L' : 'USD/G';
            }

        // CALCULATIONS & SELECTIVE DOM UPDATES
        function recalculateAll() {
            let lowestNetPrice = Infinity;
            let lowestRowId = null;

            const calculations = rows.map(row => {
                const p = parseFloat(row.price);
                const region = taxRates[row.regionKey];

                if (isNaN(p) || p <= 0 || !region) {
                    return { id: row.id, netUsdPerGal: null };
                }

                let netUsdPerGal = 0;

                if (region.type === 'US') {
                    // US State calculation: Pump Price (USD/Gal) - IFTA Tax
                    netUsdPerGal = p - region.ifta;
                } else {
                    // Canadian Province calculation:
                    // 1. Remove GST/HST sales tax from gross pump price
                    const netCadPerLiterExSalesTax = p / (1 + (region.salesTaxRate || 0));
                    // 2. Subtract IFTA provincial road tax (CAD/L)
                    const netCadPerLiter = netCadPerLiterExSalesTax - region.ifta;
                    // 3. Convert CAD to USD
                    const netUsdPerLiter = netCadPerLiter * exchangeRateCADtoUSD;
                    // 4. Convert Liters to US Gallons
                    netUsdPerGal = netUsdPerLiter * LITERS_PER_GALLON;
                }

                if (netUsdPerGal < lowestNetPrice) {
                    lowestNetPrice = netUsdPerGal;
                    lowestRowId = row.id;
                }

                return { id: row.id, netUsdPerGal };
            });

            // Selective Update to text spans WITHOUT re-rendering input elements
            calculations.forEach(calc => {
                const valSpan = document.getElementById(`net-val-${calc.id}`);
                const badge = document.getElementById(`net-badge-${calc.id}`);

                if (!valSpan || !badge) return;

                if (calc.netUsdPerGal === null) {
                    valSpan.textContent = "$0.000";
                    valSpan.className = "text-base md:text-lg font-extrabold tracking-tight text-slate-400 dark:text-slate-500";
                    badge.className = "inline-flex flex-col items-end px-3 py-1.5 rounded-lg transition-all border border-transparent";
                } else {
                    valSpan.textContent = `$${calc.netUsdPerGal.toFixed(3)}`;
                    
                    if (calc.id === lowestRowId && rows.filter(r => parseFloat(r.price) > 0).length > 1) {
                        valSpan.className = "text-base md:text-lg font-extrabold tracking-tight text-emerald-700 dark:text-emerald-300";
                        badge.className = "inline-flex flex-col items-end px-3 py-1.5 rounded-lg transition-all bg-emerald-100/80 dark:bg-emerald-950/60 border border-emerald-300 dark:border-emerald-700 shadow-sm";
                    } else {
                        valSpan.className = "text-base md:text-lg font-extrabold tracking-tight text-slate-800 dark:text-slate-100";
                        badge.className = "inline-flex flex-col items-end px-3 py-1.5 rounded-lg transition-all bg-slate-100 dark:bg-slate-700/50 border border-slate-200 dark:border-slate-600";
                    }
                }
            });
        }

        // TAX MODAL MANAGEMENT
        function openTaxModal() {
            const modal = document.getElementById('taxModal');
            const body = document.getElementById('taxModalBody');
            body.innerHTML = '';

            Object.keys(taxRates).sort().forEach(key => {
                const item = taxRates[key];
                const div = document.createElement('div');
                div.className = "p-3 bg-slate-50 dark:bg-slate-900 rounded-lg border border-slate-200 dark:border-slate-700 flex flex-col sm:flex-row sm:items-center justify-between gap-3";
                
                div.innerHTML = `
                    <div class="font-semibold text-slate-800 dark:text-slate-200 w-40">
                        ${item.name} (${key})
                        <span class="text-xs font-normal text-slate-400 block">${item.type === 'CA' ? 'Canada' : 'United States'}</span>
                    </div>
                    <div class="flex items-center gap-3">
                        <div>
                            <label class="text-[10px] text-slate-400 block">IFTA Rate (${item.unit})</label>
                            <input type="number" step="0.001" value="${item.ifta}" data-key="${key}" data-field="ifta" class="tax-input w-24 px-2 py-1 text-xs bg-white dark:bg-slate-800 border border-slate-300 dark:border-slate-600 rounded">
                        </div>
                        ${item.type === 'CA' ? `
                        <div>
                            <label class="text-[10px] text-slate-400 block">GST/HST Sales Tax</label>
                            <input type="number" step="0.01" value="${item.salesTaxRate}" data-key="${key}" data-field="salesTaxRate" class="tax-input w-20 px-2 py-1 text-xs bg-white dark:bg-slate-800 border border-slate-300 dark:border-slate-600 rounded">
                        </div>
                        ` : ''}
                    </div>
                `;
                body.appendChild(div);
            });

            // Input handlers for tax modal
            body.querySelectorAll('.tax-input').forEach(inp => {
                inp.addEventListener('input', (e) => {
                    const key = e.target.dataset.key;
                    const field = e.target.dataset.field;
                    const val = parseFloat(e.target.value);
                    if (!isNaN(val)) {
                        taxRates[key][field] = val;
                        recalculateAll();
                    }
                });
            });

            modal.classList.remove('hidden');
        }

        function closeTaxModal() {
            document.getElementById('taxModal').classList.add('hidden');
        }
    </script>
